**Goal:** Add an on-disk `ManifestArchive` that retains the last N installed manifests, expose `GET /versions` and `POST /select` for operator-driven rollback, and migrate `AppState` to a double-Arc pattern so a swap-in-progress never blocks in-flight reads.

```rust
//! On-disk archive of installed `ChunkManifest`s.
//!
//! Layout (all relative to `archive_dir`):
//!
//! ```text
//! <archive_dir>/
//!     a1b2c3d4e5f6a7b8.json      ← one file per installed manifest
//!     0123456789abcdef.json
//!     current  ───────────────►  a1b2c3d4e5f6a7b8.json    (symlink)
//! ```
//!
//! * Writes are temp-file + atomic `rename(2)`.
//! * `install()` is idempotent: an existing `<build_id>.json` is left untouched
//!   and reported with `is_current = false`.
//! * `install()` calls an internal `prune()` *after* writing so the most recent
//!   `retention` entries (by file mtime) are kept.

use std::fs;
use std::io::Write as _;
use std::path::{Path, PathBuf};
use std::time::{SystemTime, UNIX_EPOCH};

use anyhow::{anyhow, Context, Result};
use serde::Serialize;
use cloudpack_graph::ChunkManifest;

// ---------------------------------------------------------------------------
// Public types
// ---------------------------------------------------------------------------

/// Metadata about a single archived manifest.
#[derive(Debug, Clone, Serialize)]
pub struct ArchiveEntry {
    /// The manifest's `build_id` (16-char hex when produced by
    /// `cloudpack_pipeline::build_id::compute_build_id`).
    pub build_id: String,

    /// Wall-clock time the file was created/replaced, in ms since the Unix
    /// epoch (derived from the file's mtime).
    pub installed_at_ms: u64,

    /// On-disk size of the JSON file, in bytes.
    pub bytes: u64,

    /// `true` iff `<archive_dir>/current` resolves to this entry's file.
    pub is_current: bool,
}

/// On-disk manifest archive.
///
/// Cheap to construct (just stores `PathBuf` + `usize`). The struct is meant
/// to be wrapped in `Arc` and shared between handlers — methods take `&self`.
#[derive(Debug, Clone)]
pub struct ManifestArchive {
    /// Directory containing `<build_id>.json` files and the `current`
    /// symlink. Created if missing.
    pub archive_dir: PathBuf,

    /// Maximum number of archive entries to retain. The most recent N entries
    /// (by file mtime) survive `prune()`; older entries are deleted.
    pub retention: usize,
}

// ---------------------------------------------------------------------------
// Public API
// ---------------------------------------------------------------------------

impl ManifestArchive {
    /// Open (or create) the archive at `dir`.
    ///
    /// `retention` must be at least 1; values below 1 are clamped to 1 to
    /// guarantee the `current` entry is never pruned out from under callers.
    pub fn open(dir: &Path, retention: usize) -> Result<Self> {
        fs::create_dir_all(dir)
            .with_context(|| format!("failed to create archive dir {}", dir.display()))?;
        Ok(Self {
            archive_dir: dir.to_path_buf(),
            retention: retention.max(1),
        })
    }

    /// Install `manifest` into the archive and return its [`ArchiveEntry`].
    ///
    /// * If `<archive_dir>/<build_id>.json` already exists, the file is left
    ///   untouched and the existing entry's metadata is returned (idempotent).
    /// * Writes go through a temp file + `rename(2)` so partial writes are
    ///   impossible.
    /// * After writing (or short-circuiting), `prune()` runs to enforce
    ///   `retention`.
    pub fn install(&self, manifest: &ChunkManifest) -> Result<ArchiveEntry> {
        let build_id = manifest.build_id.clone();
        let target = self.path_for(&build_id);

        let already_present = target.exists();
        if !already_present {
            let bytes = serde_json::to_vec(manifest).context("serialize manifest to JSON")?;
            write_atomic(&target, &bytes).with_context(|| {
                format!("failed to write archive entry {}", target.display())
            })?;
        }

        // Prune *after* writing so we never delete the file we just made.
        self.prune().context("prune after install")?;

        let mut entry = entry_for(&target, &build_id)?;
        entry.is_current = self.current()?.as_deref() == Some(build_id.as_str());
        Ok(entry)
    }

    /// List all entries in `<archive_dir>`, sorted newest-first by mtime.
    pub fn list(&self) -> Result<Vec<ArchiveEntry>> {
        let current = self.current()?;
        let mut entries = read_dir_entries(&self.archive_dir, current.as_deref())?;
        entries.sort_by(|a, b| b.installed_at_ms.cmp(&a.installed_at_ms));
        Ok(entries)
    }

    /// Resolve `<archive_dir>/current` to a `build_id`, or `None` if no
    /// `current` symlink exists yet.
    pub fn current(&self) -> Result<Option<String>> {
        let link = self.archive_dir.join("current");
        match fs::read_link(&link) {
            Ok(target) => {
                let file_name = target
                    .file_name()
                    .and_then(|s| s.to_str())
                    .ok_or_else(|| anyhow!("current symlink target has no file name"))?;
                let build_id = file_name
                    .strip_suffix(".json")
                    .ok_or_else(|| anyhow!("current symlink target lacks .json suffix"))?;
                Ok(Some(build_id.to_string()))
            }
            Err(e) if e.kind() == std::io::ErrorKind::NotFound => Ok(None),
            Err(e) => Err(e).context(format!("read_link {}", link.display())),
        }
    }

    // -----------------------------------------------------------------------
    // Internal helpers
    // -----------------------------------------------------------------------

    /// Path of the JSON file backing `build_id`.
    fn path_for(&self, build_id: &str) -> PathBuf {
        self.archive_dir.join(format!("{build_id}.json"))
    }

    /// Delete archive entries beyond `retention`. The most recent `retention`
    /// files (by mtime) survive. The currently-active build (per the `current`
    /// symlink) is *always* retained even if its mtime would otherwise place
    /// it outside the window.
    fn prune(&self) -> Result<()> {
        let current = self.current()?;
        let mut entries = read_dir_entries(&self.archive_dir, current.as_deref())?;
        // Newest first.
        entries.sort_by(|a, b| b.installed_at_ms.cmp(&a.installed_at_ms));

        if entries.len() <= self.retention {
            return Ok(());
        }

        // Anything past the retention window is a deletion candidate; skip the
        // current build even if it would have been pruned.
        for stale in entries.iter().skip(self.retention) {
            if stale.is_current {
                continue;
            }
            let path = self.path_for(&stale.build_id);
            if let Err(e) = fs::remove_file(&path) {
                if e.kind() != std::io::ErrorKind::NotFound {
                    return Err(e)
                        .with_context(|| format!("prune: failed to remove {}", path.display()));
                }
            }
        }
        Ok(())
    }
}

// ---------------------------------------------------------------------------
// Free helpers (used by impl above and Task 2)
// ---------------------------------------------------------------------------

/// Write `bytes` to `path` atomically: write a sibling temp file, then
/// `rename(2)` into place. The temp file lives in the same directory as
/// `path` so the rename is guaranteed to be on the same filesystem.
pub(crate) fn write_atomic(path: &Path, bytes: &[u8]) -> Result<()> {
    let parent = path
        .parent()
        .ok_or_else(|| anyhow!("path {} has no parent", path.display()))?;
    let mut tmp = tempfile::Builder::new()
        .prefix(".tmp-")
        .suffix(".json")
        .tempfile_in(parent)
        .context("create temp file in archive dir")?;
    tmp.write_all(bytes).context("write temp file bytes")?;
    tmp.as_file().sync_all().context("fsync temp file")?;
    tmp.persist(path)
        .map_err(|e| anyhow!("persist temp file to {}: {}", path.display(), e.error))?;
    Ok(())
}

/// Read `dir`, returning one `ArchiveEntry` per `*.json` file. The `current`
/// argument is the build_id resolved from the `current` symlink (if any) so
/// every entry's `is_current` flag can be set in one pass.
pub(crate) fn read_dir_entries(dir: &Path, current: Option<&str>) -> Result<Vec<ArchiveEntry>> {
    let mut out = Vec::new();
    let read = match fs::read_dir(dir) {
        Ok(r) => r,
        Err(e) if e.kind() == std::io::ErrorKind::NotFound => return Ok(out),
        Err(e) => return Err(e).with_context(|| format!("read_dir {}", dir.display())),
    };

    for dent in read {
        let dent = dent.context("read_dir entry")?;
        let path = dent.path();
        let file_name = match path.file_name().and_then(|s| s.to_str()) {
            Some(n) => n,
            None => continue,
        };
        // Skip the `current` symlink and any temp files (".tmp-*.json").
        if file_name == "current" || file_name.starts_with(".tmp-") {
            continue;
        }
        let build_id = match file_name.strip_suffix(".json") {
            Some(b) => b,
            None => continue,
        };
        let mut entry = entry_for(&path, build_id)?;
        entry.is_current = current == Some(build_id);
        out.push(entry);
    }
    Ok(out)
}

/// Stat `path` and produce an `ArchiveEntry` (with `is_current = false`).
pub(crate) fn entry_for(path: &Path, build_id: &str) -> Result<ArchiveEntry> {
    let meta = fs::metadata(path)
        .with_context(|| format!("stat archive entry {}", path.display()))?;
    let mtime = meta
        .modified()
        .with_context(|| format!("mtime of {}", path.display()))?;
    let installed_at_ms = mtime
        .duration_since(UNIX_EPOCH)
        .unwrap_or_default()
        .as_millis() as u64;
    Ok(ArchiveEntry {
        build_id: build_id.to_string(),
        installed_at_ms,
        bytes: meta.len(),
        is_current: false,
    })
}

    ///
    /// Returns an error if the file is missing or its bytes are not a
    /// well-formed `ChunkManifest`.
    pub fn load(&self, build_id: &str) -> Result<ChunkManifest> {
        let path = self.path_for(build_id);
        let bytes = fs::read(&path)
            .with_context(|| format!("read archive entry {}", path.display()))?;
        let manifest: ChunkManifest = serde_json::from_slice(&bytes)
            .with_context(|| format!("parse archive entry {}", path.display()))?;
        Ok(manifest)

    /// Point `<archive_dir>/current` at `<build_id>.json` atomically.
    ///
    /// Implementation: create a sibling symlink at `.current.tmp-<rand>`, then
    /// `rename(2)` it over `current`. The rename is atomic on POSIX, so a
    /// concurrent reader sees either the old target or the new target — never
    /// a missing or partial link.
    ///
    /// Fails if `<build_id>.json` does not exist in the archive.
    pub fn set_current(&self, build_id: &str) -> Result<()> {
        let target = self.path_for(build_id);
        if !target.exists() {
            return Err(anyhow!(
                "set_current: build_id {build_id} is not present in archive {}",
                self.archive_dir.display()
            ));
        }

        let link = self.archive_dir.join("current");
        // Symlink target is the *relative* filename so the archive dir stays
        // portable (moving the dir doesn't break the link).
        let rel_target = format!("{build_id}.json");

        let tmp_name = format!(".current.tmp-{}", uuid::Uuid::new_v4());
        let tmp_link = self.archive_dir.join(&tmp_name);

        #[cfg(unix)]
        std::os::unix::fs::symlink(&rel_target, &tmp_link).with_context(|| {
            format!("create temp symlink {} -> {}", tmp_link.display(), rel_target)
        })?;
        #[cfg(windows)]
        std::os::windows::fs::symlink_file(&rel_target, &tmp_link).with_context(|| {
            format!("create temp symlink {} -> {}", tmp_link.display(), rel_target)
        })?;

        fs::rename(&tmp_link, &link).with_context(|| {
            format!(
                "rename {} -> {}",
                tmp_link.display(),
                link.display()
            )
        })?;
        Ok(())
    }

```rust
//! Application state for the Asset Bundling Server.
//!
//! [`AppState`] wraps the current [`ChunkManifest`] in
//! `Arc<RwLock<Arc<ChunkManifest>>>` so that:
//!
//! * **Readers** clone the inner `Arc` under a short read guard and drop the
//!   lock immediately — `snapshot()` is O(1) and never blocks.
//! * **Writers** swap the inner `Arc` atomically under a write guard. A
//!   separate `reload_lock` (`tokio::sync::Mutex`) serializes concurrent
//!   swap attempts so two operators racing on `/select` produce a clean
//!   "already in progress" error rather than corrupting state.

use std::path::Path;
use std::sync::Arc;

use anyhow::{Context, Result};
use ed25519_dalek::Signature;
use serde::Serialize;
use tokio::sync::{Mutex, RwLock};
use cloudpack_graph::ChunkManifest;

use crate::archive::ManifestArchive;
use crate::signing::ManifestVerifier;

/// Result of a successful [`AppState::swap_to`] / [`AppState::reload_from_current`].
#[derive(Debug, Clone, Serialize)]
pub struct SwapReport {
    /// `build_id` of the manifest that was active immediately before the swap.
    pub previous: String,
    /// `build_id` of the manifest now active after the swap.
    pub current: String,
}

/// Shared application state carried by every Axum handler.
///
/// All fields are cheaply `Clone`-able: cloning an `AppState` shares the same
/// underlying data via `Arc`.
#[derive(Clone)]
pub struct AppState {
    /// Current manifest, behind a double-`Arc` so readers don't block writers.
    pub manifest: Arc<RwLock<Arc<ChunkManifest>>>,

    /// On-disk archive of installed manifests; survives restarts.
    pub archive: Arc<ManifestArchive>,

    /// Serializes concurrent `/select` calls. A second swap attempt while one
    /// is in progress observes `try_lock() == Err(_)` and the handler returns
    /// `409 Conflict`.
    pub reload_lock: Arc<Mutex<()>>,

    /// Base URL of the CDN that serves chunk assets.
    pub cdn_base_url: Arc<String>,

    /// How long clients should cache a manifest response, in seconds.
    pub ttl_seconds: u64,
}

impl AppState {
    /// Load a [`ChunkManifest`] from a JSON file on disk and wrap it in
    /// [`AppState`], optionally verifying a cryptographic signature.
    ///
    /// The loaded manifest is installed into `archive` and `archive.current`
    /// is set to its `build_id` so a subsequent restart with the same archive
    /// resumes at the same build.
    pub async fn load_signed_from_disk(
        manifest_path: &Path,
        cdn_base_url: String,
        ttl_seconds: u64,
        archive: ManifestArchive,
        verify: Option<(&ManifestVerifier, &Signature)>,
    ) -> Result<Self> {
        let bytes = tokio::fs::read(manifest_path)
            .await
            .with_context(|| format!("failed to read manifest from {}", manifest_path.display()))?;

        let manifest: ChunkManifest = serde_json::from_slice(&bytes).with_context(|| {
            format!(
                "failed to parse manifest JSON from {}",
                manifest_path.display()
            )
        })?;

        if let Some((verifier, sig)) = verify {
            verifier
                .verify(&manifest, sig)
                .context("manifest signature verification failed")?;
        }

        let build_id = manifest.build_id.clone();
        archive.install(&manifest).context("seed archive")?;
        archive
            .set_current(&build_id)
            .context("point archive `current` at seed manifest")?;

        Ok(Self {
            manifest: Arc::new(RwLock::new(Arc::new(manifest))),
            archive: Arc::new(archive),
            reload_lock: Arc::new(Mutex::new(())),
            cdn_base_url: Arc::new(cdn_base_url),
            ttl_seconds,
        })
    }

    /// Convenience wrapper around [`Self::load_signed_from_disk`] that skips
    /// signature verification.
    pub async fn load_from_disk(
        manifest_path: &Path,
        cdn_base_url: String,
        ttl_seconds: u64,
        archive: ManifestArchive,
    ) -> Result<Self> {
        Self::load_signed_from_disk(manifest_path, cdn_base_url, ttl_seconds, archive, None).await
    }

    /// Return an `Arc<ChunkManifest>` snapshot of the current manifest.
    ///
    /// O(1): acquires the read guard, clones the inner `Arc`, drops the
    /// guard. Holding the returned `Arc` never blocks a subsequent swap.
    pub async fn snapshot(&self) -> Arc<ChunkManifest> {
        let guard = self.manifest.read().await;
        Arc::clone(&*guard)
    }

    /// Swap the active manifest to the archive entry for `build_id`.
    ///
    /// Sequence:
    /// 1. Acquire `reload_lock` (serializes with other swap attempts).
    /// 2. `archive.load(build_id)` — fails fast if not in archive.
    /// 3. Validate the loaded manifest's `build_id` actually matches the
    ///    requested ID (sanity check against on-disk tampering).
    /// 4. `archive.set_current(build_id)` — atomic symlink swap.
    /// 5. Write the new manifest into `self.manifest` under a write guard.
    /// 6. Return `SwapReport { previous, current }`.
    pub async fn swap_to(&self, build_id: &str) -> Result<SwapReport> {
        let _lock = self.reload_lock.lock().await;

        let new_manifest = self
            .archive
            .load(build_id)
            .with_context(|| format!("swap_to: load build_id={build_id}"))?;

        if new_manifest.build_id != build_id {
            return Err(anyhow::anyhow!(
                "swap_to: archive file for {build_id} has mismatched build_id={}",
                new_manifest.build_id
            ));
        }

        self.archive
            .set_current(build_id)
            .with_context(|| format!("swap_to: set_current {build_id}"))?;

        let new_arc = Arc::new(new_manifest);
        let mut guard = self.manifest.write().await;
        let previous = guard.build_id.clone();
        *guard = new_arc;
        drop(guard);

        Ok(SwapReport {
            previous,
            current: build_id.to_string(),
        })
    }

    /// Re-read `archive.current` from disk and swap to whatever it points at.
    ///
    /// Used by `run()` on startup if a `current` symlink already exists from
    /// a previous process — and by future file-watcher integrations.
    pub async fn reload_from_current(&self) -> Result<SwapReport> {
        let current = self
            .archive
            .current()
            .context("read archive `current` symlink")?
            .ok_or_else(|| anyhow::anyhow!("archive has no `current` symlink"))?;
        self.swap_to(&current).await
    }
}

```rust
/// `POST /select` — atomically switch the active manifest to an archived
/// `build_id`.
///
/// Operator-only endpoint. Loopback-enforced when `ConnectInfo` is present
/// (same rule as `POST /reload`).
///
/// Body: `{ "build_id": "<16-hex>" }`
/// * `200 OK`         — `{"previous":"...","current":"..."}`
/// * `404 Not Found`  — `build_id` is not present in the archive
/// * `409 Conflict`   — another swap is already in progress (`reload_lock`
///                       is held)
/// * `403 Forbidden`  — request came from a non-loopback address
/// * `500`            — unexpected I/O / archive error
async fn post_select(
    MaybeConnectAddr(addr): MaybeConnectAddr,
    State(state): State<RouterState>,
    Json(body): Json<SelectRequest>,
) -> Response {
    // Loopback enforcement (identical to /reload).
    if let Some(addr) = addr {
        if !addr.ip().is_loopback() {
            return (
                StatusCode::FORBIDDEN,
                Json(serde_json::json!({
                    "error": "select is only permitted from loopback addresses"
                })),
            )
                .into_response();
        }
    }

    // 404 — fail fast before touching the lock so a missing build_id never
    // blocks an in-flight swap from completing.
    let archive_entries = match state.app.archive.list() {
        Ok(e) => e,
        Err(e) => {
            return (
                StatusCode::INTERNAL_SERVER_ERROR,
                Json(serde_json::json!({
                    "error": format!("failed to list archive: {e}")
                })),
            )
                .into_response();
        }
    };
    if !archive_entries.iter().any(|e| e.build_id == body.build_id) {
        return (
            StatusCode::NOT_FOUND,
            Json(serde_json::json!({
                "error": format!("build_id {} is not present in archive", body.build_id)
            })),
        )
            .into_response();
    }

    // 409 — try_lock fails iff another swap is in progress. We DO NOT hold
    // this lock ourselves; swap_to acquires it internally. The try_lock here
    // is purely a fast-path "is someone else swapping right now?" check.
    match state.app.reload_lock.try_lock() {
        Ok(probe) => drop(probe),
        Err(_) => {
            return (
                StatusCode::CONFLICT,
                Json(serde_json::json!({
                    "error": "another manifest swap is already in progress"
                })),
            )
                .into_response();
        }
    }

    // Perform the swap. swap_to() re-acquires reload_lock internally; the
    // window between our try_lock probe above and swap_to's lock is small
    // (microseconds) — if a racer slips in, swap_to serializes correctly.
    match state.app.swap_to(&body.build_id).await {
        Ok(report) => (
            StatusCode::OK,
            Json(SwapResponseBody {
                previous: report.previous,
                current: report.current,
            }),
        )
            .into_response(),
        Err(e) => (
            StatusCode::INTERNAL_SERVER_ERROR,
            Json(serde_json::json!({
                "error": format!("failed to swap: {e}")
            })),
        )
            .into_response(),
    }
}
